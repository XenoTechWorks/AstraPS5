cmake_minimum_required(VERSION 3.16)
project(AstraPS5 VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(astra_diagnostics STATIC src/diagnostics.cpp)
target_include_directories(astra_diagnostics PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(astra_diagnostics PRIVATE -Wall -Wextra -Wpedantic)

add_library(astra_menu_state STATIC src/menu_state.cpp)
target_include_directories(astra_menu_state PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(astra_menu_state PRIVATE -Wall -Wextra -Wpedantic)

# The minimal graphical shell is a real native C++ window, not a web mockup.
# The CLI and tests still compile when X11 development headers aren't installed.
option(ASTRA_WITH_X11 "Build the experimental X11 menu" ON)
if(ASTRA_WITH_X11)
  find_package(X11 QUIET)
endif()
if(ASTRA_WITH_X11 AND X11_FOUND)
  add_library(astra_ui STATIC src/menu_x11.cpp)
  target_link_libraries(astra_ui PUBLIC astra_diagnostics astra_menu_state X11::X11)
  message(STATUS "Astra native UI: enabled (X11)")
else()
  add_library(astra_ui STATIC src/menu_stub.cpp)
  target_include_directories(astra_ui PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  message(STATUS "Astra native UI: disabled (install X11 development headers to enable)")
endif()
target_include_directories(astra_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(astra_ui PRIVATE -Wall -Wextra -Wpedantic)

add_executable(astra src/main.cpp)
target_link_libraries(astra PRIVATE astra_diagnostics astra_ui)
target_compile_options(astra PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
if(BUILD_TESTING)
  add_executable(astra_tests tests/diagnostics_test.cpp)
  target_link_libraries(astra_tests PRIVATE astra_diagnostics)
  add_executable(astra_menu_tests tests/menu_state_test.cpp)
  target_link_libraries(astra_menu_tests PRIVATE astra_menu_state)
  add_test(NAME diagnostics_unit COMMAND astra_tests)
  add_test(NAME menu_state_unit COMMAND astra_menu_tests)
  add_test(NAME cli_version COMMAND astra version)
  add_test(NAME cli_doctor COMMAND astra doctor)
  add_test(NAME cli_doctor_json COMMAND astra doctor --json)
  add_test(NAME cli_help COMMAND astra help)
endif()
