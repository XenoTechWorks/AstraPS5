cmake_minimum_required(VERSION 3.16)
project(AstraPS5 VERSION 0.2.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(astra_diagnostics STATIC src/diagnostics.cpp)
target_include_directories(astra_diagnostics PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(astra_diagnostics PRIVATE -Wall -Wextra -Wpedantic)

find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(astra_boot STATIC src/boot.cpp)
target_include_directories(astra_boot PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(astra_boot PUBLIC astra_diagnostics OpenSSL::Crypto)
target_compile_options(astra_boot PRIVATE -Wall -Wextra -Wpedantic)

# Linux-native + Flatpak game launcher. No shell, Steam, Proton, or PS5 integration.
add_library(astra_game STATIC src/game.cpp)
target_include_directories(astra_game PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(astra_game PUBLIC astra_diagnostics)
target_compile_options(astra_game PRIVATE -Wall -Wextra -Wpedantic)

# Steam manifests / Proton discovery. Read-only; no Steam game execution yet.
add_library(astra_steam STATIC src/steam.cpp)
target_include_directories(astra_steam PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(astra_steam PUBLIC astra_diagnostics)
target_compile_options(astra_steam PRIVATE -Wall -Wextra -Wpedantic)

# Flatpak desktop-export discovery, read-only, with strict Game-category filter.
add_library(astra_library STATIC src/library.cpp)
target_include_directories(astra_library PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(astra_library PUBLIC astra_game astra_diagnostics)
target_compile_options(astra_library PRIVATE -Wall -Wextra -Wpedantic)

add_library(astra_menu_state STATIC src/menu_state.cpp)
target_include_directories(astra_menu_state PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(astra_menu_state PRIVATE -Wall -Wextra -Wpedantic)

# The minimal graphical shell is a real native C++ window, not a web mockup.
# The CLI and tests still compile when X11 development headers aren't installed.
option(ASTRA_WITH_X11 "Build the experimental X11 menu" ON)
if(ASTRA_WITH_X11)
  find_package(X11 QUIET)
endif()
if(ASTRA_WITH_X11 AND X11_FOUND)
  add_library(astra_ui STATIC src/menu_x11.cpp)
  target_link_libraries(astra_ui PUBLIC astra_diagnostics astra_game astra_library astra_steam astra_menu_state X11::X11)
  message(STATUS "Astra native UI: enabled (X11)")
else()
  add_library(astra_ui STATIC src/menu_stub.cpp)
  target_include_directories(astra_ui PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  message(STATUS "Astra native UI: disabled (install X11 development headers to enable)")
endif()
target_include_directories(astra_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(astra_ui PRIVATE -Wall -Wextra -Wpedantic)

add_executable(astra src/main.cpp)
target_link_libraries(astra PRIVATE astra_diagnostics astra_ui astra_boot astra_game astra_library astra_steam)
target_compile_options(astra PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)

  target_link_libraries(astra_tests PRIVATE astra_diagnostics)
